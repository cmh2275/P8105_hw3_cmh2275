``` r
fullnhanes_df |>

  geom_point(alpha = 0.5)+
  geom_smooth(se = FALSE, method = "lm")+
  facet_wrap(~education)+
  labs(
    title = "Total Activity by Age and Sex Across Education Levels", 
    x = "Age (years)", 
    y = "Total Activity in MIMS",
```
